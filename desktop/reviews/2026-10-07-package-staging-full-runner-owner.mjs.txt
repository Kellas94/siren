import {readFile,writeFile,mkdir,readdir,lstat} from 'node:fs/promises';
import {spawn} from 'node:child_process';
import {createHash} from 'node:crypto';
import {resolve} from 'node:path';
import assert from 'node:assert/strict';
const baseCommit=process.argv[2];assert.match(baseCommit??'',/^[a-f0-9]{40}$/);
const root=resolve('evidence/workspace-surface/package-staging-full-'+new Date().toISOString().replaceAll(':','-'));await mkdir(root);
const paths=['package.json','package-lock.json','baseline/R78.html','../.github/workflows/desktop-verify.yml','native/generated/process-identity.exe','native/generated/process-identity.json'];
async function walk(directory){for(const entry of await readdir(directory,{withFileTypes:true})){const path=directory+'/'+entry.name;assert.ok(!entry.isSymbolicLink(),'Input link refused');if(entry.isDirectory())await walk(path);else{assert.ok(entry.isFile(),'Regular input required');paths.push(path);}}}
for(const directory of ['src','build','scripts','tests','generated'])await walk(directory);paths.sort();
const capture=async()=>Object.fromEntries(await Promise.all(paths.map(async path=>[path,createHash('sha256').update(await readFile(path)).digest('hex')])));
const before=await capture();await writeFile(root+'/before.json',JSON.stringify(before,null,2),{flag:'wx'});
const identity=['process-identity-deadline.test.mjs','process-identity.test.mjs','process-reader-native.test.mjs','fixture-temp-alias.test.mjs'];
const others=(await readdir('tests')).filter(name=>name.endsWith('.test.mjs')&&!identity.includes(name)).sort();
const results=[];
for(const [name,args] of [['identity',['--test','--test-reporter=tap','--test-concurrency=1',...identity.map(n=>'tests/'+n)]],['units',['--test','--test-reporter=tap',...others.map(n=>'tests/'+n)]]]){
 const child=spawn(process.execPath,args,{windowsHide:true,stdio:['ignore','pipe','pipe']});let output='';
 child.stdout.on('data',bytes=>output+=bytes);child.stderr.on('data',bytes=>output+=bytes);
 const exit=await new Promise((yes,no)=>{child.once('error',no);child.once('exit',(code,signal)=>yes({code,signal}));});
 await writeFile(root+'/'+name+'.log',output,{flag:'wx'});
 const counts=Object.fromEntries([...output.matchAll(/^# (tests|pass|fail|cancelled|skipped|todo) (\d+)\r?$/gm)].map(match=>[match[1],Number(match[2])]));
 results.push({name,args,...exit,counts});console.log(JSON.stringify(results.at(-1)));
}
const after=await capture(),changedInputs=paths.filter(path=>before[path]!==after[path]);
const complete=results.every(result=>result.code===0&&result.signal===null&&Number.isSafeInteger(result.counts.tests)&&result.counts.tests>0&&result.counts.pass===result.counts.tests&&result.counts.fail===0&&result.counts.cancelled===0&&result.counts.skipped===0&&result.counts.todo===0)&&changedInputs.length===0;
const result={author:'/root',baseCommit,workingTree:true,stage:'Owned package staging lifetime fix; real test builder with non-executable fixture runtime, not native application qualification',root,status:complete?'COMPLETE':'ADVERSE',results,before,after,capturedInputs:paths.length,changedInputs};
await writeFile(root+'/result.json',JSON.stringify(result,null,2),{flag:'wx'});console.log(JSON.stringify({root,status:result.status,tests:results.reduce((sum,r)=>sum+(r.counts.tests??0),0),capturedInputs:paths.length,changedInputs}));process.exitCode=complete?0:1;
