import assert from 'node:assert/strict';
import {mkdir,mkdtemp,realpath,lstat,writeFile,readFile,symlink,rename,rm} from 'node:fs/promises';
import {join,resolve,relative,isAbsolute} from 'node:path';
import {tmpdir} from 'node:os';
import {spawn} from 'node:child_process';
import {withPackageStaging} from '../../../scripts/package-staging.mjs';

const canonicalTemp=await realpath(tmpdir());
const root=await realpath(await mkdtemp(join(canonicalTemp,'siren-independent-staging-')));
const outcomes=[];
async function run(name,fn){
 try{await fn();outcomes.push({name,status:'PASS'});}
 catch(error){outcomes.push({name,status:'FAIL',error:String(error),stack:error.stack});}
}
async function fixture(name){const parent=join(root,name);await mkdir(parent);const dist=join(parent,'dist');await mkdir(dist);return {parent,dist};}
async function lockFile(path){
 // The literal path is passed as a child environment value, never shell text.
 const child=spawn('C:/Windows/System32/WindowsPowerShell/v1.0/powershell.exe',['-NoProfile','-NonInteractive','-Command',"$reviewLock = [System.IO.File]::Open($env:SIREN_REVIEW_LOCK_PATH,[System.IO.FileMode]::Open,[System.IO.FileAccess]::ReadWrite,[System.IO.FileShare]::None); try { [Console]::Out.WriteLine('LOCK_READY'); [Console]::Out.Flush(); [Console]::In.ReadLine() | Out-Null } finally { $reviewLock.Dispose() }"],{env:{...process.env,SIREN_REVIEW_LOCK_PATH:path},windowsHide:true,stdio:['pipe','pipe','pipe']});
 let output='',errors='';child.stderr.on('data',data=>errors+=data);
 await new Promise((accept,reject)=>{
  const timer=setTimeout(()=>{child.stdin.end('\n');reject(Error('Lock helper timeout: '+errors));},10000);
  child.stdout.on('data',data=>{output+=data;if(output.includes('LOCK_READY')){clearTimeout(timer);accept();}});
  child.once('error',error=>{clearTimeout(timer);reject(error);});
  child.once('exit',code=>{if(!output.includes('LOCK_READY')){clearTimeout(timer);reject(Error('Lock helper exited '+code+': '+errors));}});
 });
 return async()=>{await new Promise((accept,reject)=>{child.once('exit',code=>code===0?accept():reject(Error('Lock helper exit '+code+': '+errors)));child.stdin.end('\n');});};
}
try{
 await run('ancestor junction rejected before callback',async()=>{
  const {parent,dist}=await fixture('ancestor');const alias=join(root,'ancestor-alias');await symlink(parent,alias,'junction');let invoked=false;
  await assert.rejects(withPackageStaging(join(alias,'dist'),async()=>{invoked=true;}),/path refused/i);assert.equal(invoked,false);assert.ok((await lstat(dist)).isDirectory());
 });
 await run('replacement parent junction refuses cleanup and preserves both trees',async()=>{
  const {parent,dist}=await fixture('parent-junction');const moved=join(parent,'moved-dist'),target=join(parent,'target');await mkdir(target);await writeFile(join(target,'sentinel'),'keep');let source;
  await assert.rejects(withPackageStaging(dist,async path=>{source=path;await writeFile(join(path,'input'),'original');await rename(dist,moved);await symlink(target,dist,'junction');}),error=>error instanceof AggregateError&&error.message==='PACKAGE_STAGING_CLEANUP_FAILED');
  assert.equal(await readFile(join(target,'sentinel'),'utf8'),'keep');assert.equal(await readFile(join(moved,relative(dist,source),'input'),'utf8'),'original');
 });
 for(const fail of [false,true])await run('real OS lock cleanup failure '+(fail?'retains falsy build error':'rejects successful build'),async()=>{
  const {dist}=await fixture(fail?'lock-failed-build':'lock-successful-build');let source,unlock,observed;
  try{
   await withPackageStaging(dist,async path=>{source=path;const locked=join(path,'locked-input');await writeFile(locked,'retained diagnostic');unlock=await lockFile(locked);if(fail)throw undefined;return 'built';}).then(()=>assert.fail('Cleanup error was swallowed'),error=>observed=error);
   assert.ok(observed instanceof AggregateError);assert.equal(observed.message,'PACKAGE_STAGING_CLEANUP_FAILED');assert.equal(observed.stagingPath,source);assert.equal(observed.errors.length,fail?2:1);if(fail){assert.equal(observed.errors[0],undefined);assert.equal(observed.cause,undefined);}else assert.equal(observed.cause,observed.errors[0]);
   const cleanup=observed.errors.at(-1);assert.ok(['EPERM','EACCES','EBUSY'].includes(cleanup.code),cleanup.code);assert.ok((await lstat(source)).isDirectory());
   outcomes.push({name:'OS lock observed error '+(fail?'failed build':'successful build'),status:'INFO',code:cleanup.code});
  }finally{if(unlock)await unlock();}
 });
}finally{
 assert.equal((await realpath(root)).toLowerCase(),resolve(root).toLowerCase());assert.ok(!(await lstat(root)).isSymbolicLink());
 const within=relative(canonicalTemp,root);assert.ok(within&&!within.startsWith('..')&&!isAbsolute(within));await rm(root,{recursive:true});
}
console.log(JSON.stringify({schema:1,kind:'independent-package-staging-probe',platform:process.platform,node:process.version,fixtureRoot:root,fixtureRemoved:true,outcomes,passed:outcomes.filter(x=>x.status==='PASS').length,failed:outcomes.filter(x=>x.status==='FAIL').length},null,2));
if(outcomes.some(x=>x.status==='FAIL'))process.exitCode=1;
