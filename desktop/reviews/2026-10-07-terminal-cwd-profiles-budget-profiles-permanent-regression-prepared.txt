
// P2 regression: refuse excessive ancestry BEFORE materializing prefixes.
// Real pure module in a64MiB worker; no filesystem/dialog/native execution.
test('profiles bounds path prefix expansion before malformed evidence can exhaust64MiB',async()=>{
 const workerSource=String.raw`
const {parentPort,workerData}=require('node:worker_threads'),{win32:path}=require('node:path');
(async()=>{
 const {kind,components,url,validAncestors}=workerData,selected='C:\\'+Array(components).fill('a').join('\\');
 const chain=p=>{const root=path.parse(p).root,parts=p.slice(root.length).split('\\').filter(Boolean);return [root,...parts.map((_,i)=>path.join(root,...parts.slice(0,i+1)))].map((p,i)=>({path:p,identity:'directory-'+i,directory:true,reparse:false}));};
 let observed;
 if(kind==='cwd'){
  const {CwdAuthority}=await import(url),grant={windowId:'w',projectId:'p',epoch:1,role:'workspace'};
  const inspect=p=>{const parents=chain(p);return {canonicalPath:p,identity:parents.at(-1).identity,directory:true,reparse:false,ancestors:parents};};
  const a=new CwdAuthority({protectedRoots:['D:\\Protected'],authorize:()=>true,captureAdmission:()=>({isCurrent:()=>true}),pickDirectory:()=>({canceled:false,filePaths:[selected]}),inspectDirectory:p=>p===selected&&!validAncestors?{canonicalPath:p,identity:'selected',directory:true,reparse:false,ancestors:[]}:inspect(p)});
  try{await a.pick(grant);observed={code:'GRANTED'};}catch(e){observed={code:e.code,message:e.message};}
 }else{
  const {createShellProfileCatalogue}=await import(url),a=createShellProfileCatalogue({authorize:()=>true,captureAdmission:()=>({isCurrent:()=>true}),getSystemDirectory:()=>selected,inspectExecutable:p=>({canonicalPath:p,identity:'file',file:true,reparse:false,ancestors:validAncestors?chain(path.dirname(p)):[]}),readEnvironment:()=>({SystemRoot:'C:\\Windows',PATH:'C:\\Windows\\System32'})});
  try{await a.resolveShellProfile('powershell');observed={code:'RESOLVED'};}catch(e){observed={code:e.code,message:e.message};}
 }
 parentPort.postMessage({components,pathLength:selected.length,...observed});
})().catch(e=>parentPort.postMessage({unexpected:String(e)}));`;
 const run=(components,validAncestors)=>new Promise(done=>{
  let settled=false,observed=[];const w=new Worker(workerSource,{eval:true,workerData:{kind:'profiles',components,validAncestors,url:new URL('../src/terminal/profiles.mjs',import.meta.url).href},resourceLimits:{maxOldGenerationSizeMb:64,maxYoungGenerationSizeMb:8}});
  const finish=value=>{if(settled)return;settled=true;clearTimeout(timer);done({...value,observed});};
  const timer=setTimeout(()=>{void w.terminate();finish({status:'DEADLINE'});},10000);
  w.on('message',m=>observed.push(m));w.on('error',e=>finish({status:'WORKER_ERROR',code:e.code}));w.on('exit',exitCode=>finish({status:'EXIT',exitCode}));
 });
 const cases=[[3,false],[8192,false],[125,true],[126,true]],expected=["PROFILE_UNAVAILABLE","PROFILE_UNAVAILABLE","RESOLVED","PROFILE_UNAVAILABLE"];
 for(let i=0;i<cases.length;i++){const result=await run(...cases[i]);assert.equal(result.status,'EXIT',JSON.stringify(result));assert.equal(result.exitCode,0);assert.equal(result.observed.length,1);assert.equal(result.observed[0].code,expected[i],JSON.stringify(result));}
});
